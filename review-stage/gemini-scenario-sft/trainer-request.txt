import gc
import hashlib
import json
import time
import sys
from importlib.metadata import version
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM, BitsAndBytesConfig, set_seed, get_cosine_schedule_with_warmup, GenerationConfig
from peft import get_peft_model, LoraConfig, prepare_model_for_kbit_training, PeftModel

BASE = ROOT.parents[2] / "models/Qwen2.5-3B-Instruct"
SEEDS = [42,43,44]
tok = AutoTokenizer.from_pretrained(BASE,local_files_only=True)
tok.pad_token = tok.eos_token

def read(split):
    return json.loads((DATA/(split+".json")).read_text(encoding="utf8"))

def messages(r):
    return [{"role":"system","content":SYSTEM},{"role":"user","content":r["question"]}]

def tokens(r):
    chat = messages(r)
    prefix = tok.apply_chat_template(chat, tokenize=True, add_generation_prompt=True,return_dict=False)
    full = tok.apply_chat_template(chat+[{"role":"assistant","content":r["answer"]}],tokenize=True,return_dict=False)
    assert full[:len(prefix)]==prefix
    return full, [-100]*len(prefix)+full[len(prefix):]

def base():
    return AutoModelForCausalLM.from_pretrained(BASE,local_files_only=True,device_map={"":0},dtype=torch.bfloat16,attn_implementation="sdpa",quantization_config=BitsAndBytesConfig(load_in_4bit=True,bnb_4bit_use_double_quant=True,bnb_4bit_compute_dtype=torch.bfloat16,bnb_4bit_quant_type="nf4"))

def train(arm, seed, out):
    set_seed(seed)
    m = prepare_model_for_kbit_training(base(),use_gradient_checkpointing=True,gradient_checkpointing_kwargs={"use_reentrant":False})
    m = get_peft_model(m,LoraConfig(r=16,lora_alpha=32,lora_dropout=.05,target_modules="all-linear",task_type="CAUSAL_LM"))
    m.config.use_cache=False
    m.train()
    encoded = list(map(tokens,read(arm)))
    steps = 2*len(encoded)//4
    opt = torch.optim.AdamW((p for p in m.parameters() if p.requires_grad),lr=7e-5,weight_decay=0.)
    sched = get_cosine_schedule_with_warmup(opt,2,steps)
    gen = torch.Generator().manual_seed(seed)
    log=[]
    start=time.monotonic()
    torch.cuda.reset_peak_memory_stats()
    for epoch in range(2):
        order = torch.randperm(len(encoded),generator=gen).tolist()
        for i in range(0,len(order),4):
            batch=[encoded[k] for k in order[i:i+4]]
            width=max(len(a) for a,b in batch)
            ids=torch.tensor([a+[tok.pad_token_id]*(width-len(a)) for a,b in batch],device="cuda")
            mask=torch.tensor([[1]*len(a)+[0]*(width-len(a)) for a,b in batch],device="cuda")
            labels=torch.tensor([b+[-100]*(width-len(b)) for a,b in batch],device="cuda")
            loss=m(input_ids=ids,attention_mask=mask,labels=labels).loss
            if not torch.isfinite(loss): raise ValueError("Nonfinite training loss")
            loss.backward()
            torch.nn.utils.clip_grad_norm_(m.parameters(),1.)
            opt.step()
            sched.step()
            opt.zero_grad(set_to_none=True)
            log.append(loss.item())
    m.save_pretrained(out/"adapter")
    save(out/"training.json",dict(seed=seed,steps=steps,loss=log,elapsed=time.monotonic()-start,peak_gib=torch.cuda.max_memory_allocated()/2**30,input_tokens=sum(len(x) for x,y in encoded),supervised_tokens=sum(sum(v!=-100 for v in y) for x,y in encoded)))
    print("Trained",arm,seed,flush=True)

def evaluate(out,adapter,budget=96):
    m=base()
    if adapter: m=PeftModel.from_pretrained(m,adapter)
    tok.padding_side="left"
    m.eval()
    cases=read("near")+read("far")
    predictions=[]
    cfg=GenerationConfig(max_new_tokens=budget,do_sample=False,eos_token_id=m.generation_config.eos_token_id,pad_token_id=tok.pad_token_id,use_cache=True)
    ends=cfg.eos_token_id if isinstance(cfg.eos_token_id,list) else [cfg.eos_token_id]
    with torch.inference_mode():
        for i in range(0,len(cases),12):
            chunk=cases[i:i+12]
            enc=tok.apply_chat_template([messages(r) for r in chunk],padding=True,tokenize=True,add_generation_prompt=True,return_dict=True,return_tensors="pt").to("cuda")
            new=m.generate(**enc,generation_config=cfg)[:,enc.input_ids.shape[1]:].tolist()
            for r,ids in zip(chunk,new):
                predictions.append(dict(id=r["id"],text=tok.decode(ids,skip_special_tokens=True).strip(),capped=not any(k in ends for k in ids)))
    save(out/"predictions.json",predictions)
    print("Evaluated",out.name,len(predictions),flush=True)
